# ری‌استارت کامل
cd ~/groupbot && pkill -9 -f "main.py" ; sleep 3 && nohup bash bot_watchdog.sh > ~/groupbot_watchdog.log 2>&1 & sleep 15 && tail -n 10 ~/groupbot_bot.log

# لاگ زنده
tail -F ~/groupbot_bot.log

# جستجو توی کد
grep -rn "کلیدواژه" ~/groupbot/modules/*.py | grep -v ".bak"

# کامپایل
python3 -m py_compile ~/groupbot/modules/file.py

# بکاپ قبل تغییر
cp ~/groupbot/modules/file.py ~/groupbot/modules/file.py.bak_$(date +%H%M%S)

# دیدن بکاپ‌ها
ls -lht ~/groupbot/modules/file.py.bak_* | head -5

# دیاگنوستیک
cd ~/groupbot && python3 ~/diag.py

# ارسال دیاگ به تلگرام
cd ~/groupbot && python3 ~/diag_send.py
